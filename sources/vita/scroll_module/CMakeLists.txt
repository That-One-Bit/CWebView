cmake_minimum_required(VERSION 3.16)
project(CwvScroll C)
if(NOT DEFINED ENV{VITASDK})
  message(FATAL_ERROR "VITASDK is not set")
endif()
include($ENV{VITASDK}/share/vita.cmake REQUIRED)
add_executable(CwvScroll src/main.c)
target_compile_options(CwvScroll PRIVATE -O2 -Wall -Wextra -std=c11)
target_link_libraries(CwvScroll m)
vita_create_self(CwvScroll.suprx CwvScroll UNSAFE CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/exports.yml)
vita_create_stubs(CwvScroll-stubs CwvScroll ${CMAKE_CURRENT_SOURCE_DIR}/exports.yml)
add_custom_target(CwvScroll-package DEPENDS CwvScroll.suprx-self CwvScroll-stubs)
